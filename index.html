# Odoo 18 Enterprise - Student Ledger & Kiosk Model
from odoo import models, fields, api, _
from odoo.exceptions import ValidationError

class StudentTuitionPayment(models.Model):
    _name = 'lu.student.tuition.payment'
    _inherit = ['mail.thread', 'mail.activity.mixin']
    _description = 'Lusail Campus Automated Kiosk Payment'

    student_id = fields.Many2one('lu.student', string='Student', required=True, index=True)
    kiosk_id = fields.Char(string='Terminal Hardware ID', readonly=True)
    amount_paid = fields.Monetary(string='Amount (QAR)', currency_field='currency_id')
    currency_id = fields.Many2one('res.currency', default=lambda self: self.env.company.currency_id)
    payment_method = fields.Selection([
        ('qpay', 'Qatar Debit Card'),
        ('apple_pay', 'NFC Contactless')
    ], required=True)
    state = fields.Selection([
        ('draft', 'Initiated'),
        ('authorized', 'Gateway Validated'),
        ('posted', 'Ledger Synchronized')
    ], default='draft', tracking=True)

    @api.model
    def process_kiosk_transaction(self, payload):
        """Atomic RPC endpoint invoked by campus payment kiosks"""
        student = self.env['lu.student'].search([
            ('national_id', '=', payload.get('national_id'))
        ], limit=1)
        if not student:
            raise ValidationError(_("Student profile not registered."))
        
        record = self.create({
            'student_id': student.id,
            'kiosk_id': payload.get('kiosk_mac'),
            'amount_paid': payload.get('amount'),
            'payment_method': payload.get('method'),
            'state': 'posted'
        })
        # Synchronize with Ellucian Banner SIS ledger
        record.action_sync_banner_ledger()
        return {'status': 'success', 'ref': record.id, 'new_balance': student.balance}
